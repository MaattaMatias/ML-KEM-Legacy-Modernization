Prompt: generate tests to measure the ML-KEM performance using the same metrics from before.

This benchmarking suite isolates the ML-KEM-1024 operations that occur between the Edge Gateway and the Cloud.
import time
import tracemalloc
import sys
import oqs

def run_mlkem_baselines():
    print("\n" + "="*65)
    print(" POST-QUANTUM (ML-KEM-1024) PERFORMANCE BASELINES")
    print("="*65)

    # 1. Setup - Generate the Cloud's Keypair
    kem_server = oqs.KeyEncapsulation("ML-KEM-1024")
    public_key = kem_server.generate_keypair()
    private_key = kem_server.export_secret_key()
    
    kem_client = oqs.KeyEncapsulation("ML-KEM-1024")
    
    # 2. Network / Payload Bloat Measurement
    print("\n--- 1. Network / Payload Metrics (WAN Link) ---")
    
    # Encapsulate to get the ciphertext and shared secret
    ciphertext, shared_secret = kem_client.encap_secret(public_key)
    
    pk_size = len(public_key)
    ct_size = len(ciphertext)
    ss_size = len(shared_secret)
    
    print(f"Public Key Size (Sent Cloud -> Gateway):    {pk_size} bytes")
    print(f"Ciphertext Size (Sent Gateway -> Cloud):    {ct_size} bytes")
    print(f"Negotiated Shared Secret (Used for AES):    {ss_size} bytes")
    print("Note: ML-KEM requires sending relatively large keys and ciphertexts ")
    print("compared to traditional Elliptic Curve (ECDH), which impacts WAN bandwidth.")

    # 3. Memory Allocation (Edge Gateway RAM Proxy)
    print("\n--- 2. Memory Metrics (Gateway RAM Proxy) ---")
    
    # Measure Encapsulation (Gateway side)
    tracemalloc.start()
    _, _ = kem_client.encap_secret(public_key)
    _, peak_encap = tracemalloc.get_traced_memory()
    tracemalloc.stop()
    
    # Measure Decapsulation (Cloud side)
    kem_server_decaps = oqs.KeyEncapsulation("ML-KEM-1024")
    kem_server_decaps.secret_key = private_key
    tracemalloc.start()
    _ = kem_server_decaps.decaps_secret(ciphertext)
    _, peak_decap = tracemalloc.get_traced_memory()
    tracemalloc.stop()

    print(f"Peak RAM allocated during Encapsulation (Gateway): {peak_encap} bytes")
    print(f"Peak RAM allocated during Decapsulation (Cloud):   {peak_decap} bytes")
    print("Context: Easily handled by an Edge Gateway (e.g., Raspberry Pi with GBs of RAM), ")
    print("but would be impossible on the 8-bit Arduino's 2KB SRAM.")

    # 4. Latency / CPU Time Measurement
    print("\n--- 3. Latency Metrics ---")
    
    iterations = 100
    
    # Measure Encapsulation Time
    encap_times = []
    for _ in range(iterations):
        start_time = time.perf_counter()
        kem_client.encap_secret(public_key)
        encap_times.append((time.perf_counter() - start_time) * 1000)
    
    avg_encap_time = sum(encap_times) / len(encap_times)
    
    # Measure Decapsulation Time
    decap_times = []
    for _ in range(iterations):
        start_time = time.perf_counter()
        kem_server_decaps.decaps_secret(ciphertext)
        decap_times.append((time.perf_counter() - start_time) * 1000)
        
    avg_decap_time = sum(decap_times) / len(decap_times)

    print(f"Average Encapsulation Time (Gateway): {avg_encap_time:.4f} ms")
    print(f"Average Decapsulation Time (Cloud):   {avg_decap_time:.4f} ms")
    print("="*65 + "\n")

if __name__ == "__main__":
    run_mlkem_baselines()
